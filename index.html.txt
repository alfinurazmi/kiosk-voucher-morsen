  <h2>Kiosk Pemindai Voucher</h2>
  
  <div id="video-container">
    <video id="camera" autoplay playsinline muted></video>
  </div>

  <button id="snapButton" onclick="captureAndSend()">Ambil Foto & Proses</button>
  <div id="result"></div>

  <canvas id="canvas" style="display:none;"></canvas>

  <script>
    // Ganti URL di bawah ini dengan URL Web App Google Apps Script Anda
    const WEB_APP_URL = 'https://script.google.com/macros/s/AKfycbxBoAKoG4riE271KF2k32HqR9vBoEIdC4DFV-TZw_Ez-6plTjqwQE7n0VEP_N3yLb8q/exec';

    const video = document.getElementById('camera');
    const canvas = document.getElementById('canvas');
    const snapButton = document.getElementById('snapButton');
    const resultDiv = document.getElementById('result');

    // Aktifkan kamera belakang HP secara otomatis
    async function startCamera() {
      try {
        const stream = await navigator.mediaDevices.getUserMedia({
          video: { facingMode: 'environment', width: { ideal: 1280 }, height: { ideal: 720 } }
        });
        video.srcObject = stream;
      } catch (err) {
        resultDiv.className = 'error';
        resultDiv.innerText = 'Gagal mengakses kamera: Pastikan izin kamera diaktifkan pada browser Anda.';
      }
    }

    startCamera();

    function captureAndSend() {
      if (!video.srcObject) {
        alert('Kamera belum aktif!');
        return;
      }

      snapButton.disabled = true;
      snapButton.innerText = 'Memproses...';
      resultDiv.className = 'loading';
      resultDiv.innerText = 'Sedang membaca voucher dengan AI...';

      canvas.width = video.videoWidth;
      canvas.height = video.videoHeight;
      const ctx = canvas.getContext('2d');
      ctx.drawImage(video, 0, 0, canvas.width, canvas.height);

      const dataUrl = canvas.toDataURL('image/jpeg', 0.8);

      fetch(WEB_APP_URL, {
        method: 'POST',
        body: JSON.stringify({ dataUrl: dataUrl })
      })
      .then(res => res.json())
      .then(data => {
        snapButton.disabled = false;
        snapButton.innerText = 'Ambil Foto & Proses';

        if (data.status === 'success') {
          resultDiv.className = 'success';
          resultDiv.innerHTML = `<b>Berhasil Disimpan!</b><br>Nama: ${data.nama}<br>No HP: ${data.no_hp}`;
        } else {
          resultDiv.className = 'error';
          resultDiv.innerText = 'Gagal: ' + data.message;
        }
      })
      .catch(err => {
        snapButton.disabled = false;
        snapButton.innerText = 'Ambil Foto & Proses';
        resultDiv.className = 'error';
        resultDiv.innerText = 'Terjadi kesalahan koneksi ke server.';
      });
    }
  </script>
